Open the main application at http://localhost:3001

You can test the toast notifications by using the useToast hook in any component:

import { useToast } from './contexts/ToastContext';

function MyComponent() {
  const { showSuccess, showError, showWarning, showInfo } = useToast();
  
  return (
    <div>
      <button onClick={() => showSuccess('Order created successfully!')}>
        Show Success Toast
      </button>
      <button onClick={() => showError('Failed to create shipment')}>
        Show Error Toast
      </button>
      <button onClick={() => showWarning('Please verify the address')}>
        Show Warning Toast
      </button>
      <button onClick={() => showInfo('Rate comparison in progress')}>
        Show Info Toast
      </button>
    </div>
  );
}

Next Steps

To integrate toasts into existing components (e.g., OrderForm, RateComparisonTable), replace error setState calls with toast notifications.

Example Integration:

// In OrderForm.jsx or any component
import { useToast } from '../contexts/ToastContext';

export default function OrderForm({ onOrderCreated }) {
  const { showSuccess, showError } = useToast();
  
  const handleSubmit = async (data) => {
    try {
      const response = await createOrder(data);
      showSuccess('Order created successfully!');
      onOrderCreated(response);
    } catch (error) {
      showError(error.message || 'Failed to create order');
    }
  };
  
  // ... rest of component
}
